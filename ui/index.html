Integrate RouteOps in 3 Lines of Code

Drop RouteOps as a smart proxy in front of OpenAI, Anthropic, or Ollama to eliminate 60%+ of API spend.

import requests

# 1. Dispatch prompt to RouteOps Gateway
decision = requests.post("http://127.0.0.1:8000/v1/route", json={"query": prompt}).json()

# 2. Skip LLM entirely on cache-hit or guardrail block
if decision["is_cache_hit"] or decision["is_guardrail_triggered"]:
    return decision["response_text"]

# 3. Route sanitized prompt to target tier (SLM, RAG, or Frontier)
llm_client.execute(model=decision["routed_tier"], prompt=decision["redacted_query"])
import { routeQuery } from "@routeops/client";

// 1. Triage incoming user prompt
const decision = await routeQuery({ query: prompt, endpoint: "http://127.0.0.1:8000/v1/route" });

// 2. Immediate zero-cost return on cache hit or prompt injection
if (decision.is_cache_hit || decision.is_guardrail_triggered) {
  return decision.response_text;
}

// 3. Route to matched LLM engine using sanitized query
return llm[decision.routed_tier].generate(decision.redacted_query);
